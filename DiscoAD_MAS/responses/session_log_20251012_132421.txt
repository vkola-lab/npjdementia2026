# Define outcome variable: Dementia (binary: 1 = dementia, 0 otherwise)

# Chi-square test for association between age group and dementia

print(f"\nChi-Square Test Results (Age Group vs. Dementia):")
print(f"Chi² = {chi2_stat:.4f}, p-value = {p_val:.4f}")

# If p < 0.05, we reject null hypothesis — there is an association between age group and dementia

# Alternative: Logistic regression to model dementia risk as function of age

from sklearn.preprocessing import LabelEncoder

# Prepare features and target
