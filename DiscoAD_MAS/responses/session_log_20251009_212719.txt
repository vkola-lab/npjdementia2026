import pandas as pd
import numpy as np
from scipy import stats
import matplotlib.pyplot as plt
import seaborn as sns

# Load the filtered dataset from the working directory
file_path = "./working/filtered_dataset.csv"
try:
    df = pd.read_csv(file_path)
    print(f"Dataset loaded successfully. Shape: {df.shape}")

    print("Error: The file './working/filtered_dataset.csv' was not found.")

except Exception as e:

# Display basic information about the dataset
print("\n--- Dataset Information ---")

print("\n--- Basic Statistics ---")
print(df.describe())

# Check for missing values
missing_values = df.isnull().sum()
print("\n--- Missing Values per Column ---")
print(missing_values[missing_values > 0])

# Focus on variables relevant to the hypothesis: age and dementia incidence
# We'll examine the relationship between Subjects_age_at_visit and Met_criteria_for_dementia

# Filter out rows where either age or dementia status is missing
valid_data = df.dropna(subset=['Subjects_age_at_visit', 'Met_criteria_for_dementia'])

print(f"\nAfter removing rows with missing values in age or dementia status: {len(valid_data)} observations")

# Descriptive statistics for age group by dementia status
age_by_dementia = valid_data.groupby('Met_criteria_for_dementia')['Subjects_age_at_visit'].agg([
    'mean', 'median', 'std', 'count'
]).round(2)

print("\n--- Age Statistics by Dementia Status (0 = No, 1 = Yes) ---")
print(age_by_dementia)

# Perform t-test to assess whether mean age differs significantly between those with and without dementia
try:
    # Separate the two groups
    ages_no_dementia = valid_data[valid_data['Met_criteria_for_dementia'] == 0]['Subjects_age_at_visit']
    ages_with_dementia = valid_data[valid_data['Met_criteria_for_dementia'] == 1]['Subjects_age_at_visit']

    # Test for normality of age distributions (Shapiro-Wilk test)
    shapiro_no, p_no = stats.shapiro(ages_no_dementia)
    shapiro_yes, p_yes = stats.shapiro(ages_with_dementia)

    print(f"\nNormality Test Results:")
    print(f"Age (No Dementia): Shapiro-Wilk p-value = {p_no:.4f}")
    print(f"Age (With Dementia): Shapiro-Wilk p-value = {p_yes:.4f}")

    # If both groups are normally distributed, use parametric t-test; otherwise, use Mann-Whitney U
    if p_no > 0.05 and p_yes > 0.05:
        print("Both age groups appear normally distributed. Using independent samples t-test.")
        t_stat, p_value_t = stats.ttest_ind(ages_no_dementia, ages_with_dementia, equal_var=False)
        
        print(f"\nIndependent Samples t-test result:")
        print(f"T-statistic: {t_stat:.4f}, P-value: {p_value_t:.4f}")
        
        # Effect size using Cohen's d
        diff_mean = abs(ages_with_dementia.mean() - ages_no_dementia.mean())
        pooled_std = np.sqrt((ages_with_dementia.std()**2 + ages_no_dementia.std()**2) / 2)
        cohen_d = diff_mean / pooled_std
        print(f"Cohen's d (effect size): {cohen_d:.3f}")
        
        # Interpretation
        if cohen_d < 0.2:
            effect_interpretation = "Small"
        elif cohen_d < 0.5:
            effect_interpretation = "Moderate"
        else:
            effect_interpretation = "Large"
        print(f"Effect size interpretation: {effect_interpretation}")
        
    else:
        print("At least one group deviates from normality. Using Mann-Whitney U test.")
        u_stat, p_value_u = stats.mannwhitneyu(ages_no_dementia, ages_with_dementia, alternative='two-sided')
        print(f"\nMann-Whitney U test result:")
        print(f"U-statistic: {u_stat:.4f}, P-value: {p_value_u:.4f}")
        
        # Effect size using r (rank correlation)
        r = stats.spearmanr(ages_no_dementia, ages_with_dementia).correlation
        print(f"Spearman rank correlation (r): {r:.4f}")
        
except Exception as e:
    print(f"Error during statistical test: {e}")

# Perform t-test to assess whether mean age differs significantly between those with and without dementia
try:
    # Separate the two groups
    ages_no_dementia = valid_data[valid_data['Met_criteria_for_dementia'] == 0]['Subjects_age_at_visit']
    ages_with_dementia = valid_data[valid_data['Met_criteria_for_dementia'] == 1]['Subjects_age_at_visit']

    # Test for normality of age distributions (Shapiro-Wilk test)
    shapiro_no, p_no = stats.shapiro(ages_no_dementia)
    shapiro_yes, p_yes = stats.shapiro(ages_with_dementia)

    print(f"\nNormality Test Results:")
    print(f"Age (No Dementia): Shapiro-Wilk p-value = {p_no:.4f}")
    print(f"Age (With Dementia): Shapiro-Wilk p-value = {p_yes:.4f}")

    # If both groups are normally distributed, use parametric t-test; otherwise, use Mann-Whitney U
    if p_no > 0.05 and p_yes > 0.05:
        print("Both age groups appear normally distributed. Using independent samples t-test.")
        t_stat, p_value_t = stats.ttest_ind(ages_no_dementia, ages_with_dementia, equal_var=False)
        
        print(f"\nIndependent Samples t-test result:")
        print(f"T-statistic: {t_stat:.4f}, P-value: {p_value_t:.4f}")
        
        # Effect size using Cohen's d
        diff_mean = abs(ages_with_dementia.mean() - ages_no_dementia.mean())
        pooled_std = np.sqrt((ages_with_dementia.std()**2 + ages_no_dementia.std()**2) / 2)
        cohen_d = diff_mean / pooled_std
        print(f"Cohen's d (effect size): {cohen_d:.3f}")
        
        # Interpretation
        if cohen_d < 0.2:
            effect_interpretation = "Small"

            effect_interpretation = "Moderate"
