import pandas as pd
import numpy as np
from scipy.stats import pearsonr

# Load the filtered dataset
file_path = "./working/filtered_Verbal_fluency_is_positively_associated_with_seman_NACC.csv"
try:
    df = pd.read_csv(file_path)
except FileNotFoundError:
    print("Error: The file was not found at the specified path.")
    exit()

# Check for missing values
print("Missing values per column:")
print(df.isnull().sum())

# Select relevant columns
relevant_columns = ["Number_of_correct_Fwords_generated_in_1_minute", 
                    "Number_of_correct_Lwords_generated_in_1_minute", 
                    "Memory"]

# Filter out rows with missing values in relevant columns
df_filtered = df[relevant_columns].dropna()

# Perform Pearson correlation between verbal fluency (Fwords and Lwords) and semantic memory
print("\nCorrelation between Number_of_correct_Fwords_generated_in_1_minute and Memory:")
corr_f_words_memory, p_value_f_words_memory = pearsonr(
    df_filtered["Number_of_correct_Fwords_generated_in_1_minute"], 
    df_filtered["Memory"]
)
print(f"Correlation coefficient: {corr_f_words_memory:.3f}, p-value: {p_value_f_words_memory:.4f}")

print("\nCorrelation between Number_of_correct_Lwords_generated_in_1_minute and Memory:")
corr_l_words_memory, p_value_l_words_memory = pearsonr(
    df_filtered["Number_of_correct_Lwords_generated_in_1_minute"], 
    df_filtered["Memory"]
)
print(f"Correlation coefficient: {corr_l_words_memory:.3f}, p-value: {p_value_l_words_memory:.4f}")

# Summary of results
print("\nHypothesis validation summary:")
if (abs(corr_f_words_memory) >= 0.3 and p_value_f_words_memory < 0.05):
    print("The positive association between F-word fluency and semantic memory is statistically significant.")
else:
    print("No strong evidence for a positive association between F-word fluency and semantic memory.")

if (abs(corr_l_words_memory) >= 0.3 and p_value_l_words_memory < 0.05):
    print("The positive association between L-word fluency and semantic memory is statistically significant.")
else:
    print("No strong evidence for a positive association between L-word fluency and semantic memory.")

import pandas as pd
import numpy as np
from scipy.stats import pearsonr

# Load the filtered dataset
file_path = "./working/filtered_Individuals_with_stronger_semantic_memory_exhibit_NACC.csv"
try:
    df = pd.read_csv(file_path)
except FileNotFoundError:
    print("Error: The file was not found at the specified path.")
    exit()

# Check for missing values
print("Missing values per column:")
print(df.isnull().sum())

# Select relevant columns
relevant_columns = ["Number_of_correct_Fwords_generated_in_1_minute", "Memory", "Subjects_age_at_visit"]

# Filter out rows with missing values in relevant columns
df_filtered = df[relevant_columns].dropna()

print("\nCorrelation between Memory and Number_of_correct_Fwords_generated_in_1_minute:")
corr_memory_f_words, p_value_memory_f_words = pearsonr(
    df_filtered["Memory"], 
    df_filtered["Number_of_correct_Fwords_generated_in_1_minute"]
)
print(f"Correlation coefficient: {corr_memory_f_words:.3f}, p-value: {p_value_memory_f_words:.4f}")

# Assess if the correlation meets criteria for a meaningful positive association
if abs(corr_memory_f_words) >= 0.3 and p_value_memory_f_words < 0.05:
    print("There is a statistically significant and moderately strong positive association between semantic memory and F-word fluency.")
else:
    print("No strong evidence for a positive association between semantic memory and F-word fluency.")

# Optional: Check for possible influence of age (as a potential confounder)
print("\nAge distribution across memory groups (mean ± std):")
age_by_memory_group = df_filtered.groupby(pd.cut(df_filtered["Memory"], bins=3, labels=["Low", "Medium", "High"])).agg({"Subjects_age_at_visit": ["mean", "std"]}).round(2)
print(age_by_memory_group)
