import pandas as pd
import numpy as np
from scipy import stats
import matplotlib.pyplot as plt
import seaborn as sns

# Load the filtered dataset from the working directory
file_path = "./working/filtered_dataset.csv"
try:
    df = pd.read_csv(file_path)
    print(f"Dataset loaded successfully. Shape: {df.shape}")
except FileNotFoundError:
    print("Error: The file './working/filtered_dataset.csv' was not found.")
    exit()
except Exception as e:
    print(f"Error loading the dataset: {e}")
    exit()

# Display basic information about the dataframe
print("\n--- Data Info ---")
print(df.info())

print("\n--- Basic Statistics ---")
print(df.describe())

# Check for missing values
missing_values = df.isnull().sum()
print("\n--- Missing Values per Column ---")
print(missing_values[missing_values > 0])

# Focus on the key variables for the hypothesis
# We're testing: Higher age is associated with increased dementia incidence
# Relevant columns:
#   - Subjects_age_at_visit (age)
#   - Met_criteria_for_dementia (dementia status: 0 = no, 1 = yes)

# Remove rows with missing values in the two key variables we are analyzing
df_filtered = df.dropna(subset=['Subjects_age_at_visit', 'Met_criteria_for_dementia'])

print(f"\nAfter removing missing values in age and dementia status: {df_filtered.shape[0]} rows remain.")

# Descriptive statistics for age group by dementia status
print("\n--- Age Summary by Dementia Status ---")
age_by_dementia = df_filtered.groupby('Met_criteria_for_dementia')['Subjects_age_at_visit'].agg(['mean', 'std', 'count'])
print(age_by_dementia)

# Perform a t-test to assess whether higher age is associated with increased dementia incidence
# Null hypothesis: No difference in mean age between those with and without dementia
# Alternative hypothesis: Those with dementia have higher mean age

try:
    # Extract age values for those with and without dementia
    age_no_dementia = df_filtered[df_filtered['Met_criteria_for_dementia'] == 0]['Subjects_age_at_visit']
    age_with_dementia = df_filtered[df_filtered['Met_criteria_for_dementia'] == 1]['Subjects_age_at_visit']

    # Conduct independent samples t-test assuming equal variances
    t_stat, p_value = stats.ttest_ind(
        age_with_dementia, 
        age_no_dementia,
        equal_var=True,
        nan_policy='omit'
    )

    # Calculate effect size using Cohen's d
    def cohens_d(group1, group2):
        diff = abs(np.mean(group1) - np.mean(group2))
        std_pooled = np.sqrt(((np.var(group1) + np.var(group2)) / 2))
        return diff / std_pooled

    effect_size = cohens_d(age_with_dementia, age_no_dementia)

    print(f"\n--- Statistical Test Results ---")
    print(f"T-statistic: {t_stat:.4f}")
    print(f"P-value: {p_value:.4f}")
    print(f"Cohen's d (effect size): {effect_size:.3f}")

    # Interpretation
    if p_value < 0.05:
        print("Conclusion: There is statistically significant evidence that higher age is associated with increased dementia incidence.")
    else:
        print("Conclusion: No statistically significant association found between age and dementia incidence.")

    # Optional: Visualize the relationship
    plt.figure(figsize=(10, 6))
    sns.boxplot(data=df_filtered, x='Met_criteria_for_dementia', y='Subjects_age_at_visit')
    plt.title("Age Distribution by Dementia Status")
    plt.xlabel("Dementia Status (0 = No, 1 = Yes)")
    plt.ylabel("Age at Visit")
    plt.grid(True, axis='y')
    plt.tight_layout()
    plt.savefig("./working/dementia_age_boxplot.png")
    plt.show()

except Exception as e:
