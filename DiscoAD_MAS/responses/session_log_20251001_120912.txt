```python
numerical_cols = df.select_dtypes(include=[np.number]).columns.tolist()

else:
    desc_stats = df[numerical_cols].describe().T
    print(desc_stats[['mean', 'std', 'min', '25%', '50%', '75%', 'max']].round(3))

# Step 4: Check for outliers using IQR method (basic outlier detection)
print("\n=== OUTLIER DETECTION (IQR Method) FOR KEY VARIABLES ===")
key_numerical_vars = ['Subjects_age_at_visit', 'Years_of_education', 'Global_CDR', 'Memory', 'Orientation',
                      'Judgment_and_problemsolving', 'body_mass_index_BMI', 'Hippocampal_atrophy']

outlier_results = {}
for col in key_numerical_vars:
    if col not in df.columns:
        continue
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    
    outliers = df[(df[col] < lower_bound) | (df[col] > upper_bound)]
    outlier_count = len(outliers)
    
    outlier_results[col] = {
        'lower_bound': round(lower_bound, 3),
        'upper_bound': round(upper_bound, 3),
        'outlier_count': outlier_count,
        'total_samples': len(df)
    }
    
    print(f"{col}: Outliers detected = {outlier_count} ({outlier_count/len(df)*100:.2f}%)")

# Step 5: Normality testing (Shapiro-Wilk for small samples; Kolmogorov-Smirnov otherwise)

normality_results = {}

for col in key_numerical_vars:
    if col not in df.columns:
        continue
    n = len(df[col])
    if n < 30:
        try:
            stat, p_value = stats.shapiro(df[col])
            normality_results[col] = {'shapiro_wilk_stat': round(stat, 3), 'p_value': round(p_value, 4)}
            if p_value > 0.05:
                print(f"{col}: Normal (p = {p_value})")
            else:
                print(f"{col}: Not normally distributed (p = {p_value})")
        except Exception as e:
            print(f"Normality test failed for {col}: {e}")
    else:
        # For larger samples, use KS test
        try:
            stat, p_value = stats.kstest(df[col], 'norm')
            normality_results[col] = {'ks_stat': round(stat, 3), 'p_value': round(p_value, 4)}
            if p_value > 0.05:
                print(f"{col}: Normal (p = {p_value})")
            else:
                print(f"{col}: Not normally distributed (p = {p_value})")
        except Exception as e:
            print(f"KS test failed for {col}: {e}")
```

```python
import os
import pandas as pd
import numpy as np
from scipy import stats
import warnings

# Suppress warnings for cleaner output
warnings.filterwarnings("ignore")
```

```python
# Define path to the filtered dataset
data_path = "./working/filtered_dataset.csv"

# Check if the file exists before attempting to load
if not os.path.exists(data_path):
    raise FileNotFoundError(f"Dataset not found at {data_path}. Please ensure the data has been properly saved.")

# Load the dataset
try:
    df = pd.read_csv(data_path)
except Exception as e:
    raise RuntimeError(f"Error loading the dataset: {str(e)}")

# Display basic information about the dataset
print("=" * 60)
print("📊 EXPLORATORY DATA ANALYSIS (EDA)")
print("=" * 60)

print(f"✅ Dataset loaded successfully.")
print(f"📊 Shape: {df.shape}")
print(f"📁 Columns: {list(df.columns)}")
print("-" * 60)

# Check data types
print("\n📋 Data Types:")
print(df.dtypes.to_string())

print("-" * 60)

# Assess missing values
missing_summary = df.isnull().sum()
total_missing = missing_summary.sum()
print(f"\n⚠️ Missing Values Summary:")
print(missing_summary[missing_summary > 0])
print(f"\nTotal missing entries: {total_missing} ({(total_missing / df.size)*100:.2f}% of total data)")

# Identify highly missing variables (>80% missing)
high_missing_threshold = 0.8
high_missing_vars = missing_summary[missing_summary / len(df) >= high_missing_threshold]
if len(high_missing_vars) > 0:
    print(f"\n🔴 High-missing variables (>80% missing): {list(high_missing_vars.index)}")
else:
    print("\n🟢 No variables exceed 80% missingness.")

print("-" * 60)

# Descriptive statistics for numeric variables
numeric_columns = df.select_dtypes(include=[np.number]).columns.tolist()

if len(numeric_columns) == 0:
    print("❌ No numeric columns found for descriptive statistics.")
else:
    print("\n📈 Descriptive Statistics (Numeric Variables):")
    desc_stats = df[numeric_columns].describe()
    print(desc_stats.round(3))

print("-" * 60)

# Normality testing for key continuous variables (using Shapiro-Wilk where applicable)
normality_test_results = {}

key_numeric_vars = [
    'Memory', 'Orientation', 'Judgment_and_problemsolving',
    'Global_CDR', 'MoCA_Total_Score_corrected_for_education',
    'Total_MMSE_score_using_DLROW', 'Subjects_age_at_visit',
    'Years_of_education', 'Number_of_APOE_e4_alleles'
]

for col in key_numeric_vars:
    if col not in df.columns:
        continue
    data = df[col].dropna()
    if len(data) < 3:
        print(f"⚠️ Insufficient data for normality test in '{col}' (only {len(data)} non-null values). Skipping...")
        continue
    
    # Shapiro-Wilk test (valid for n < 5000)
    stat, p_value = stats.shapiro(data)
    alpha = 0.05
    is_normal = p_value > alpha
    normality_test_results[col] = {
        'statistic': round(stat, 4),
        'p_value': round(p_value, 4),
        'is_normal': is_normal,
        'sample_size': len(data)
    }
    
    print(f"{col}: Shapiro-Wilk (p={p_value:.4f}) → {'Normal' if is_normal else 'Non-normal'}")

print("-" * 60)

# Check for extreme outliers using IQR method
outliers_summary = {}
for col in key_numeric_vars:
    if col not in df.columns:
        continue
    data = df[col].dropna()
    if len(data) < 4:
        continue
        
    Q1 = data.quantile(0.25)
    Q3 = data.quantile(0.75)
    IQR = Q3 - Q1
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    
    outlier_mask = (data < lower_bound) | (data > upper_bound)
    outlier_count = outlier_mask.sum()
    if outlier_count > 0:
        outliers_summary[col] = {
            'count': outlier_count,
            'lower_bound': round(lower_bound, 2),
            'upper_bound': round(upper_bound, 2),
            'proportion': round(outlier_count / len(data), 4)
        }
        print(f"{col}: Outliers detected → {outlier_count} values outside [{round(lower_bound,2)}, {round(upper_bound,2)}]")
    else:
        print(f"{col}: No significant outliers detected.")

print("-" * 60)

# Check for presence of the target variable: Donepezil use
print("\n🔍 Target Variable: 'Reported_current_use_of_a_FDAapproved_medication_for_Alzheimers_disease_symptoms'")
if 'Reported_current_use_of_a_FDAapproved_medication_for_Alzheimers_disease_symptoms' in df.columns:
    donepezil_use = df['Reported_current_use_of_a_FDAapproved_medication_for_Alzheimers_disease_symptoms']
    print(f"✅ Column exists. Type: {donepezil_use.dtype}")
    print(f"Values: Unique counts = {donepezil_use.nunique()}, Range = [{donepezil_use.min():.2f}, {donepezil_use.max():.2f}]")
    print(f"Mean usage = {donepezil_use.mean():.3f}, Std = {donepezil_use.std():.3f}")
    
    # Check if binary (0/1) or continuous
    value_counts = donepezil_use.value_counts(dropna=False)
    print(f"Value Counts:\n{value_counts.to_dict()}")
else:
    print("❌ Critical: 'Donepezil use' column NOT found. Cannot proceed with hypothesis testing.")
```
