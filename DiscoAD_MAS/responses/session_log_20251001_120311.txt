def run_t_test(group1, group2, name):
    """Run parametric or non-parametric test depending on normality."""
    # First check for normality using Shapiro-Wilk (small samples)
    try:
        from scipy.stats import shapiro
        stat1, p1 = shapiro(group1)
        stat2, p2 = shapiro(group2)
        
        if p1 > 0.05 and p2 > 0.05:
            # Assume normal — use t-test
            t_stat, p_value = ttest_ind(group1, group2, equal_var=False)
            method = "T-test (parametric)"

import numpy as np
import warnings
from scipy import stats

    raise

# Display basic information about the dataset
print("\n=== DATA INFORMATION ===")
print(df.info())

# Check for missing values
print("\n=== MISSING VALUES ===")
missing_summary = df.isnull().sum()

# Summary statistics for numeric columns
print("\n=== DESCRIPTIVE STATISTICS FOR NUMERIC VARIABLES ===")
numeric_columns = df.select_dtypes(include=[np.number]).columns
if len(numeric_columns) == 0:

else:
    desc_stats = df[numeric_columns].describe()
    print(desc_stats)

# Key variables for hypothesis:
# - Memory: MMSE and MoCA scores
# - Cardiac arrest: Heart_attackcardiac_arrest
# - Parkinson’s disease dementia: Parkinsons_disease_PD

# Step 1: Filter for valid memory assessments
memory_cols = [
    'Total_MMSE_score_using_DLROW',
    'MoCA_Total_Score_corrected_for_education',
    'MoCA_Abstraction',
    'MoCA_Visuospatialexecutive_Clock_contour',
    'MoCA_Attention_Digits'
]

# Ensure we don't drop too many due to missingness; keep only rows with complete memory data
valid_memory_mask = df[memory_cols].notna().all(axis=1)
filtered_df = df[valid_memory_mask].copy()

print(f"\nAfter filtering missing memory data: {len(filtered_df)} observations remain.")

# Descriptive stats for memory scores
print("\n=== MEMORY SCORES AFTER FILTERING ===")
memory_desc = filtered_df[memory_cols].describe()

# Check distributions of key binary variables
binary_vars = [
    'Heart_attackcardiac_arrest',
    'Parkinsons_disease_PD'
]

print("\n=== BINARY VARIABLE DISTRIBUTIONS ===")
for col in binary_vars:

        print(f"{col}: {filtered_df[col].value_counts(dropna=False)}")

# Now define meaningful groups for hypothesis testing
# Focus on:
# Group 1: Had cardiac arrest (Heart_attackcardiac_arrest == 1)
# Group 2: No cardiac arrest (Heart_attackcardiac_arrest == 0)
# Stratify by Parkinson’s disease dementia (PD)

# Create subset with only relevant binary indicators
study_subset = filtered_df[
    (filtered_df['Heart_attackcardiac_arrest'].isin([0, 1])) &

]

    control_group = study_subset[(study_subset['Heart_attackcardiac_arrest'] == 0) &

    if len(control_group) < 2:

        # Extract memory scores for each condition

        for label, (ca, pd_flag) in groups_to_compare:

                    method = "Mann-Whitney U"
                    
            except Exception:
                u_stat, p_val = stats.mannwhitneyu(moca_group, moca_control, alternative='two-sided')
                method = "Mann-Whitney U"
            
            effect_size = abs(np.mean(moca_group) - np.mean(moca_control)) / np.std(moca_group + moca_control) * 2
            
            results[label]["moca"] = {
                "method": method,
                "t_stat": t_stat if "t_stat" in locals() else None,
                "u_stat": u_stat,
                "p_value": p_val,
                "effect_size": effect_size
            }
        
        # Print results in clean format
        print("\n=== HYPOTHESIS TEST RESULTS (vs Control Group: No CA, No PD) ===")
        print("{:<30} {:<15} {:<15} {:<15}".format("Group", "MMSE P-value", "MMSE Effect Size", "MoCA P-value"))
        
        for label, res in results.items():
            mmse_p = res["mmse"]["p_value"]
            moca_p = res["moca"]["p_value"]
            mmse_eff = res["mmse"]["effect_size"]
            moca_eff = res["moca"]["effect_size"]
            
            print(f"{label:<30} {mmse_p:<15.4f} {mmse_eff:<15.3f} {moca_p:<15.4f}")

        # Final interpretation
        significant_groups = [k for k, v in results.items() if v["mmse"]["p_value"] < 0.05 or v["moca"]["p_value"] < 0.05]
        
        print("\n=== FINAL INTERPRETATION ===")
        if significant_groups:
            print("The hypothesis is supported: There are statistically significant differences in memory scores between subjects with and without cardiac arrest, particularly in the presence of Parkinson’s disease dementia. This suggests a synergistic impact of cardiovascular events and neurodegenerative pathology on cognitive decline.")
        else:
            print("No statistically significant differences were observed in memory scores across groups, suggesting that the interaction between cardiac arrest and Parkinson’s disease dementia may not be strong enough to support the hypothesis under current data. Further investigation with larger sample sizes or more refined phenotyping may be necessary.")

        # Save results for reproducibility
        results_df = pd.DataFrame({
            "Group": list(results.keys()),
            "MMSE_P_Value": [res["mmse"]["p_value"] for res in results.values()],
            "MMSE_Effect_Size": [res["mmse"]["effect_size"] for res in results.values()],
            "MoCA_P_Value": [res["moca"]["p_value"] for res in results.values()],
            "MoCA_Effect_Size": [res["moca"]["effect_size"] for res in results.values()]
        })
        results_df.to_csv("./working/hypothesis_analysis_results.csv", index=False)
        print("\nResults saved to: ./working/hypothesis_analysis_results.csv")
