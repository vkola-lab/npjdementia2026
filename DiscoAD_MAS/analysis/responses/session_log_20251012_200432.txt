def calculate_odds_ratio(table):
    """Compute odds ratio and 95% confidence interval from 2x2 contingency table"""
    a, b, c, d = table.iloc[0,0], table.iloc[0,1], table.iloc[1,0], table.iloc[1,1]

    # Odds ratio
