
- `t` = time from baseline to dementia diagnosis or death

```python
import pandas as pd
import numpy as np
from scipy import stats
import matplotlib.pyplot as plt
import seaborn as sns

# Load the processed dataset from the working directory
file_path = "./working/processed_dataset.csv"
try:
    df = pd.read_csv(file_path)
    print(f"Dataset loaded successfully. Shape: {df.shape}")
except FileNotFoundError:
    print("Error: File not found at expected path. Please ensure the dataset was properly saved.")
    raise
except Exception as e:
    print(f"Error loading dataset: {e}")
    raise

# Display basic information about the dataset
print("\n=== DATA INFORMATION ===")
print(df.info())

print("\n=== BASIC STATISTICS ===")
print(df.describe())

# Check for missing values
missing_values = df.isnull().sum()
print("\n=== MISSING VALUES ===")
print(missing_values[missing_values > 0])

# Focus on the key variables for the hypothesis: EDUC and DEMENTED
# We want to examine whether higher years of education are associated with lower dementia incidence.

# Filter out rows where DEMENTED is missing
filtered_df = df.dropna(subset=['EDUC', 'DEMENTED_YES', 'DEMENTED_NO'])

# Create binary variable for dementia status: 1 = dementia, 0 = no dementia
filtered_df['DEMENTED'] = np.where(filtered_df['DEMENTED_YES'] == 1, 1, 0)

# Verify the distribution of dementia cases vs. non-cases
print("\n=== DISTRIBUTION OF DEMENTIA STATUS ===")
dementia_distribution = filtered_df['DEMENTED'].value_counts(normalize=True)
print(dementia_distribution)

# Descriptive statistics by dementia status
print("\n=== DESCRIBED STATISTICS BY DEMENTIA STATUS ===")
stats_by_dementia = filtered_df.groupby('DEMENTED')['EDUC'].agg(['mean', 'std', 'count']).round(2)
print(stats_by_dementia)

# Perform independent samples t-test to compare average years of education between those with and without dementia
print("\n=== INDEPENDENT SAMPLES T-TEST: EDUC BY DEMENTIA STATUS ===")
t_stat, p_val = stats.ttest_ind(
    filtered_df[filtered_df['DEMENTED'] == 0]['EDUC'],
    filtered_df[filtered_df['DEMENTED'] == 1]['EDUC'],
    equal_var=False  # Use Welch's t-test since variances may differ
)

print(f"T-statistic: {t_stat:.4f}, P-value: {p_val:.4f}")

# Calculate effect size using Cohen's d
def cohens_d(group1, group2):
    diff = abs(np.mean(group1) - np.mean(group2))
    std1, std2 = np.std(group1, ddof=1), np.std(group2, ddof=1)
    n1, n2 = len(group1), len(group2)
    pooled_std = np.sqrt(((n1 - 1) * std1**2 + (n2 - 1) * std2**2) / (n1 + n2 - 2))
    return diff / pooled_std

cohens_d_value = cohens_d(
    filtered_df[filtered_df['DEMENTED'] == 0]['EDUC'],
    filtered_df[filtered_df['DEMENTED'] == 1]['EDUC']
)

print(f"Cohen's d (effect size): {cohens_d_value:.3f}")

# Optional: Visualize the relationship
plt.figure(figsize=(10, 6))
sns.boxplot(x='DEMENTED', y='EDUC', data=filtered_df)
plt.title('Years of Education by Dementia Status')
plt.xlabel('Dementia Status (0 = No, 1 = Yes)')
plt.ylabel('Years of Education')
plt.grid(True, axis='y')
plt.tight_layout()
plt.show()

# Final conclusion statement
if p_val < 0.05:
    print("\n✅ Hypothesis supported: There is a statistically significant association between greater years of education and lower dementia incidence.")
else:
    print("\n❌ Hypothesis not supported: No statistically significant association found between years of education and dementia incidence.")

print(f"\nAdditional notes:")
print(f"- Effect size (Cohen’s d): {cohens_d_value:.3f} — moderate to large effect if >0.5")
print(f"- Sample size: {len(filtered_df)} participants analyzed")
```
