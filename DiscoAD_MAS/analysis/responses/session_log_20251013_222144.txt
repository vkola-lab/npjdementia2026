# Add confidence intervals using bootstrapping
def bootstrap_mean(data, n_bootstrap=1000):
    bootstrap_means = []
    for _ in range(n_bootstrap):
        sample = np.random.choice(data, replace=True, size=len(data))
        bootstrap_means.append(np.mean(sample))
    return np.array(bootstrap_means)

# Compute bootstrap confidence intervals for dementia incidence across education levels
