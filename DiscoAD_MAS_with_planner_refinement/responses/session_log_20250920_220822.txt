    exit()

# Validate required columns
required_columns = ["APOE_genotype", "Rey_Auditory_Verbal_Learning_total_delayed_recall"]

    if col not in df.columns:
        print(f"Error: Missing required column '{col}' in the dataset.")
        exit()

# Filter individuals based on APOE genotype (assuming ε4 allele corresponds to APOE_genotype == 4)
df['has_APOE_ε4'] = (df['APOE_genotype'] == 4).astype(int)

# Extract delayed recall scores for groups
grouped_data = df.groupby('has_APOE_ε4')['Rey_Auditory_Verbal_Learning_total_delayed_recall']

# Perform t-test to compare mean delayed recall between APOE ε4 carriers and non-carriers
group1_mean = grouped_data.get_group(1).mean()
group0_mean = grouped_data.get_group(0).mean()

t_stat, p_value = stats.ttest_ind(
    df[df['has_APOE_ε4'] == 1]['Rey_Auditory_Verbal_Learning_total_delayed_recall'],
    df[df['has_APOE_ε4'] == 0]['Rey_Auditory_Verbal_Learning_total_delayed_recall']
)

print(f"Mean delayed recall score for APOE ε4 carriers: {group1_mean:.3f}")
print(f"Mean delayed recall score for non-carriers: {group0_mean:.3f}")
print(f"T-statistic: {t_stat:.4f}, P-value: {p_value:.4e}")

# Check for missing values and outliers using IQR method
def detect_outliers(data):
    Q1 = data.quantile(0.25)
    Q3 = data.quantile(0.75)
    IQR = Q3 - Q1
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    return data[(data < lower_bound) | (data > upper_bound)]

recall_scores = df['Rey_Auditory_Verbal_Learning_total_delayed_recall']
outliers = detect_outliers(recall_scores)
if len(outliers) > 0:
    print(f"Detected {len(outliers)} potential outliers in delayed recall scores.")
else:
    print("No outliers detected in delayed recall scores.")

# Summary statistics
print("\nSummary Statistics:")
print(df[['Rey_Auditory_Verbal_Learning_total_delayed_recall']].describe())

group_no_use = df_clean[df_clean['inhibitor_binary'] == 0]['Cognitive_status_at_UDS_visit']

# Perform independent samples t-test assuming unequal variances
try:
    t_stat, p_val = stats.ttest_ind(group_use, group_no_use, equal_var=False)
    print(f"T-test result: t-statistic = {t_stat:.4f}, p-value = {p_val:.4f}")
    
    if p_val < 0.05:
        print("Conclusion: There is a statistically significant difference in cognitive outcomes between patients using and not using cholinesterase inhibitors, supporting the hypothesis that cholinesterase inhibitor use is associated with improved cognitive outcomes in patients with congestive heart failure.")
    else:
        print("Conclusion: No statistically significant difference in cognitive outcomes is observed between patients using and not using cholinesterase inhibitors, suggesting insufficient evidence to support the hypothesis.")
        
except Exception as e:
    print(f"Error during t-test: {e}")
