```python
import pandas as pd
import numpy as np
from scipy.stats import pearsonr, spearmanr
from statsmodels.formula.api import ols
import warnings
warnings.filterwarnings("ignore")

# Load the filtered dataset from the working directory
file_path = "./working/filtered_Stroke_is_associated_with_increased_levels_of_infl_NACC.csv"
try:
    df = pd.read_csv(file_path)
    print(f"Dataset loaded successfully. Shape: {df.shape}")
except FileNotFoundError:
    print("Error: The file was not found in the specified path.")
    exit()
except Exception as e:
    print(f"Error loading the dataset: {e}")
    exit()

# Confirm relevant columns are present
relevant_columns = [
    'Stroke', 
    'Reported_current_use_of_an_anticoagulant_or_antiplatelet_agent',
    'Reported_current_use_of_nonsteroidal_antiinflammatory_medication',
    'Subjects_age_at_visit',
    'Subjects_sex'
]

missing_cols = [col for col in relevant_columns if col not in df.columns]
if missing_cols:
    print(f"Missing required columns: {missing_cols}")
    exit()

# Ensure numeric types for analysis
for col in relevant_columns:
    if df[col].dtype == 'object':
        df[col] = pd.to_numeric(df[col], errors='coerce')

# Drop rows with missing critical values
df_clean = df.dropna(subset=['Stroke', 'Reported_current_use_of_nonsteroidal_antiinflammatory_medication'])

print(f"After dropping missing values: {df_clean.shape}")

# Perform Pearson and Spearman correlation between Stroke and inflammatory markers
correlations = {}
for var in ['Reported_current_use_of_nonsteroidal_antiinflammatory_medication', 
            'Reported_current_use_of_an_anticoagulant_or_antiplatelet_agent']:
    # Pearson correlation
    corr_pearson, p_pearson = pearsonr(df_clean['Stroke'], df_clean[var])
    # Spearman correlation
    corr_spearman, p_spearman = spearmanr(df_clean['Stroke'], df_clean[var])
    
    correlations[var] = {
        'Pearson': (corr_pearson, p_pearson),
        'Spearman': (corr_spearman, p_spearman)
    }
    
    print(f"\n--- Correlation between Stroke and {var} ---")
    print(f"Pearson: r = {corr_pearson:.3f}, p-value = {p_pearson:.3e}")
    print(f"Spearman: r = {corr_spearman:.3f}, p-value = {p_spearman:.3e}")

# Multivariate linear regression adjusting for age and sex
# Model: Stroke ~ NSAID use + Anticoagulant use + Age + Sex
formula = 'Stroke ~ Reported_current_use_of_nonsteroidal_antiinflammatory_medication + Reported_current_use_of_an_anticoagulant_or_antiplatelet_agent + Subjects_age_at_visit + Subjects_sex'

model = ols(formula, data=df_clean).fit()

# Summary of regression results
print("\n--- Multivariate Linear Regression Results (Adjusted) ---")
print(model.summary())

# Extract key statistics
adjusted_r_squared = model.rsquared_adj
f_statistic = model.fvalue
p_value_f = model.f_pvalue

print(f"Adjusted R-squared: {adjusted_r_squared:.3f}")
print(f"F-statistic: {f_statistic:.3f}, p-value: {p_value_f:.3e}")

# Check for multicollinearity (VIF would require additional library, but we check correlation matrix here)
corr_matrix = df_clean[['Reported_current_use_of_nonsteroidal_antiinflammatory_medication',
                        'Reported_current_use_of_an_anticoagulant_or_antiplatelet_agent',
                        'Subjects_age_at_visit',
                        'Subjects_sex']].corr()
print("\n--- Pairwise Correlation Matrix (among predictors) ---")
print(corr_matrix.round(3))
```
