    df_filtered['Years_of_education'], 
    df_filtered['Hypercholesterolemia']

df_filtered['Education_Group'] = pd.cut(

    bins=[0, 12, 16, np.inf],

```python
import pandas as pd
import numpy as np
from scipy import stats

# Load the filtered dataset
file_path = "./working/filtered_Individuals_with_hypercholesterolemia_have_lower_l_NACC.csv"
try:
    df = pd.read_csv(file_path)
    print("Dataset loaded successfully.")
except FileNotFoundError:
    print("Error: The file was not found at the specified path.")
    exit()
except Exception as e:
    print(f"Error loading the dataset: {e}")
    exit()

# Check for necessary columns
if 'Years_of_education' not in df.columns or 'Hypercholesterolemia' not in df.columns:
    print("Error: Required columns ('Years_of_education', 'Hypercholesterolemia') missing from the dataset.")
    exit()

# Filter out rows where Hypercholesterolemia is missing or NaN
df_filtered = df.dropna(subset=['Years_of_education', 'Hypercholesterolemia'])

# Ensure years of education is non-negative
df_filtered = df_filtered[df_filtered['Years_of_education'] >= 0]

# Create binary indicator for presence of hypercholesterolemia (1 if >0, else 0)
df_filtered['Has_Hypercholesterolemia'] = (df_filtered['Hypercholesterolemia'] > 0).astype(int)

# Split data into those with and without hypercholesterolemia
with_hypert = df_filtered[df_filtered['Has_Hypercholesterolemia'] == 1]
without_hypert = df_filtered[df_filtered['Has_Hypercholesterolemia'] == 0]

# Compare average years of education between groups
mean_edu_with = np.mean(with_hypert['Years_of_education'])
mean_edu_without = np.mean(without_hypert['Years_of_education'])

print(f"Mean years of education with hypercholesterolemia: {mean_edu_with:.2f}")
print(f"Mean years of education without hypercholesterolemia: {mean_edu_without:.2f}")

# Perform independent t-test to assess significance
t_stat, p_val = stats.ttest_ind(
    with_hypert['Years_of_education'],
    without_hypert['Years_of_education'],
    equal_var=False  # Use Welch's t-test for unequal variances
)

print(f"T-statistic: {t_stat:.3f}")
print(f"P-value: {p_val:.4f}")

# Interpretation
alpha = 0.05
if p_val < alpha:
    print("Conclusion: There is a statistically significant difference in education levels between individuals with and without hypercholesterolemia.")
else:
    print("Conclusion: No statistically significant difference in education levels found between individuals with and without hypercholesterolemia.")
```
