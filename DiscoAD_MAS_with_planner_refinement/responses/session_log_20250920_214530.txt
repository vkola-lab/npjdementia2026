import numpy as np
from scipy import stats

# Load the filtered dataset
file_path = "./working/filtered_Individuals_with_diabetes_have_a_higher_prevalence_NACC.csv"

    df = pd.read_csv(file_path)
except FileNotFoundError:
    print("Error: The file was not found at the specified path.")
    exit()

print("Dataset shape:", df.shape)
print("\nFirst few rows:")
print(df.head())

print("\nMissing values per column:")
print(df.isnull().sum())

# Filter out records where 'Smoked_cigarettes_in_last_30_days' is zero (no smoking) to focus on active smokers
df_active_smokers = df[df['Smoked_cigarettes_in_last_30_days'] > 0]

# Check if we have enough active smokers for meaningful comparison
if df_active_smokers.empty:
    print("No active smokers in the dataset — unable to proceed with analysis.")
else:
    # Separate data by diabetes status
    diabetic_smokers = df_active_smokers[df_active_smokers['Diabetes'] == 1]
    non_diabetic_smokers = df_active_smokers[df_active_smokers['Diabetes'] == 0]

    # Compute average number of packs smoked per day for each group
    avg_packs_diabetic = diabetic_smokers['Average_number_of_packs_smoked_per_day'].mean()
    avg_packs_non_diabetic = non_diabetic_smokers['Average_number_of_packs_smoked_per_day'].mean()

    print(f"\nAverage packs smoked per day (diabetic): {avg_packs_diabetic:.2f}")
    print(f"Average packs smoked per day (non-diabetic): {avg_packs_non_diabetic:.2f}")

    # Perform t-test to assess statistical significance
    try:
        t_stat, p_value = stats.ttest_ind(
            diabetic_smokers['Average_number_of_packs_smoked_per_day'],
            non_diabetic_smokers['Average_number_of_packs_smoked_per_day'],
            equal_var=False
        )
        print(f"T-statistic: {t_stat:.3f}, P-value: {p_value:.4f}")
        
        # Interpretation
        alpha = 0.05
        if p_value < alpha:
            print("Conclusion: There is a statistically significant difference in average pack consumption between diabetic and non-diabetic individuals.")
        else:
            print("Conclusion: No statistically significant difference in average pack consumption between diabetic and non-diabetic individuals.")
            
    except Exception as e:
        print(f"Error during t-test: {e}")

import pandas as pd
import numpy as np
from scipy import stats

# Load the filtered dataset
file_path = "./working/filtered_Smoking_is_associated_with_an_increased_risk_of_de_NACC.csv"
try:
    df = pd.read_csv(file_path)
except FileNotFoundError:
    print("Error: The file was not found at the specified path.")
    exit()

# Display basic information about the dataset
print("Dataset shape:", df.shape)
print("\nFirst few rows:")
print(df.head())

# Check for missing values
print("\nMissing values per column:")
print(df.isnull().sum())

# Identify relevant variables
# We are testing whether smoking is associated with increased diabetes risk
# Use total years smoked as exposure, and diabetes status as outcome

# Handle missing values: exclude rows where key variables are missing
# Only keep observations with valid smoking duration and diabetes status
df_filtered = df.dropna(subset=['Diabetes', 'Total_years_smoked_cigarettes'])

# Ensure diabetes is binary (0 or 1)
df_filtered['Diabetes'] = df_filtered['Diabetes'].fillna(0).astype(int)

# Check if we have sufficient sample size
if df_filtered.empty:
    print("No valid data after filtering — unable to proceed with analysis.")
else:
    # Split into groups based on smoking exposure (use total years smoked)
    # We'll perform a t-test to compare average diabetes prevalence between smokers and non-smokers
    # Define smoker vs non-smoker: consider those who smoked for ≥1 year as "current smokers"
    df_filtered['Is_Smoker'] = df_filtered['Total_years_smoked_cigarettes'] >= 1
    
    smokers = df_filtered[df_filtered['Is_Smoker']]
    non_smokers = df_filtered[~df_filtered['Is_Smoker']]
    
    # Calculate diabetes prevalence (proportion with diabetes)
    prev_diabetes_smokers = smokers['Diabetes'].mean()
    prev_diabetes_non_smokers = non_smokers['Diabetes'].mean()
    
    print(f"\nDiabetes prevalence among smokers (≥1 year): {prev_diabetes_smokers:.3f}")
    print(f"Diabetes prevalence among non-smokers (<1 year): {prev_diabetes_non_smokers:.3f}")
    
    # Perform chi-square test for independence between smoking status and diabetes
    contingency_table = pd.crosstab(index=df_filtered['Is_Smoker'], columns=df_filtered['Diabetes'])
    try:
        chi2, p_val, dof, expected = stats.chi2_contingency(contingency_table)
        print(f"\nChi-square statistic: {chi2:.3f}, P-value: {p_val:.4f}")
        
        if p_val < 0.05:
            print("Conclusion: Smoking status is significantly associated with diabetes risk.")
        else:
            print("Conclusion: No significant association between smoking and diabetes risk.")
            
    except Exception as e:
        print(f"Error performing chi-square test: {e}")
